import pandas as pd
import matplotlib.pyplot as plt

# 加载数据
df = pd.read_csv('car_price_prediction.csv')

# 统计'manufacturer'列中每个变量的数量
manufacturer_counts = df['Manufacturer'].value_counts()

# 对结果进行排序（默认是按数量降序排序）
sorted_manufacturer_counts = manufacturer_counts.sort_values(ascending=False)

# 绘制柱状图
plt.figure(figsize=(10, 8))  # 设置画布大小
sorted_manufacturer_counts.plot(kind='bar')  # 使用 'bar' 类型绘制柱状图
plt.title('Manufacturer Distribution')  # 设置图表标题
plt.xlabel('Manufacturer')  # 设置x轴标签
plt.ylabel('Count')  # 设置y轴标签
plt.xticks(rotation=45, ha="right")  # 旋转x轴标签，以便更好地展示长名称
plt.tight_layout()  # 调整子图参数，以便子图之间不会重叠

plt.show()
